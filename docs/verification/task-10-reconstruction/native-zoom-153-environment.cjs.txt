const {chromium} = require('playwright');
const http = require('node:http');
const fs = require('node:fs');
(async()=>{
 const startedAt=new Date().toISOString();
 const server=http.createServer((req,res)=>res.end('<!doctype html><html><body style="margin:0"><main style="width:100%;font-size:16px">Actual native zoom environment probe</main></body></html>'));
 await new Promise(r=>server.listen(0,'127.0.0.1',r));
 let context;
 try{
 const ext='/workspace/scratch/e5db2ba73c1a/zoom-environment/extension';
 context=await chromium.launchPersistentContext('/workspace/scratch/e5db2ba73c1a/zoom-environment/profile',{executablePath:'/tmp/av-chromium-current/chromium',channel:'chromium',headless:true,viewport:null,timeout:15000,ignoreDefaultArgs:['--disable-extensions'],args:['--no-sandbox','--disable-dev-shm-usage','--window-size=1366,768',`--disable-extensions-except=${ext}`,`--load-extension=${ext}`]});
 console.log(JSON.stringify({startedAt,browser:context.browser()?.version()}));
 const page=context.pages()[0]??await context.newPage();const url=`http://127.0.0.1:${server.address().port}`;await page.goto(url);
 const metrics=()=>page.evaluate(()=>({width:innerWidth,height:innerHeight,outerWidth,outerHeight,dpr:devicePixelRatio,screenWidth:screen.width,cssZoom:getComputedStyle(document.documentElement).zoom,visualScale:visualViewport.scale}));
 const before=await metrics();
 const sw=context.serviceWorkers()[0]??await context.waitForEvent('serviceworker',{timeout:10000});
 const zoom=await sw.evaluate(async(url)=>{const tabs=await chrome.tabs.query({});const tab=tabs.find(t=>t.url?.startsWith(url));if(!tab)throw Error('localhost tab missing');await chrome.tabs.setZoomSettings(tab.id,{mode:'automatic',scope:'per-tab'});await chrome.tabs.setZoom(tab.id,2);return {zoom:await chrome.tabs.getZoom(tab.id),settings:await chrome.tabs.getZoomSettings(tab.id)};},url);
 await page.waitForTimeout(250);console.log(JSON.stringify({before,zoom,after:await metrics(),completedAt:new Date().toISOString(),scope:'Environment only, no Aurevane page or acceptance asserted'}));
 }finally{await context?.close();await new Promise(r=>server.close(r));}
})().catch(e=>{console.error(e.stack);process.exitCode=1;});
